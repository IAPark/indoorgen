# cloth shim: Jolt Physics (MIT) and Bullet (zlib) soft bodies behind one C entry point each (see cloth_shim.cpp)
cmake_minimum_required(VERSION 3.20)
project(clothshim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(JOLT_DIR "" CACHE PATH "JoltPhysics checkout")
set(BULLET_DIR "" CACHE PATH "bullet3 checkout")

# --- Jolt
set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(USE_ASSERTS OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
set(FLOATING_POINT_EXCEPTIONS_ENABLED OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_OBJECT_STREAM OFF CACHE BOOL "" FORCE)
set(JPH_USE_DX12 OFF CACHE BOOL "" FORCE)
set(JPH_USE_VK OFF CACHE BOOL "" FORCE)
set(JPH_USE_MTL OFF CACHE BOOL "" FORCE)
add_subdirectory(${JOLT_DIR}/Build jolt)

# --- Bullet (only the Bullet 2 libraries a soft body needs)
set(USE_DOUBLE_PRECISION OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
set(BULLET_VERSION 3.25)
set(BULLET_VERSION_MAJOR 3)
set(BULLET_VERSION_MINOR 25)
set(INSTALL_LIBS OFF)
include_directories(${BULLET_DIR}/src)
add_subdirectory(${BULLET_DIR}/src/LinearMath bt_linearmath)
add_subdirectory(${BULLET_DIR}/src/BulletCollision bt_collision)
add_subdirectory(${BULLET_DIR}/src/BulletDynamics bt_dynamics)
add_subdirectory(${BULLET_DIR}/src/BulletSoftBody bt_softbody)

add_library(clothshim STATIC cloth_shim.cpp)
target_include_directories(clothshim PRIVATE ${BULLET_DIR}/src)
target_link_libraries(clothshim PUBLIC Jolt BulletSoftBody BulletDynamics BulletCollision LinearMath)
